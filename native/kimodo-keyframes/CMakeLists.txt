cmake_minimum_required(VERSION 3.25)
project(local_character_keyframes LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(kmd-keyframes main.cpp)
foreach(file gguf ggml_weights denoiser diffusion motion_rep motion_decode llm_text_encoder llm_tokenizer)
  target_sources(kmd-keyframes PRIVATE "${KIMODO_SOURCE_DIR}/src/${file}.cpp")
endforeach()
target_include_directories(kmd-keyframes PRIVATE "${KIMODO_SOURCE_DIR}/src" "${KIMODO_SOURCE_DIR}/ggml/include")
target_compile_definitions(kmd-keyframes PRIVATE KIMODO_HAVE_GGML=1 KIMODO_HAVE_GGML_VULKAN=1 _CRT_SECURE_NO_WARNINGS)
target_link_libraries(kmd-keyframes PRIVATE "${GGML_BUILD_DIR}/src/Release/ggml.lib" "${GGML_BUILD_DIR}/src/Release/ggml-base.lib" "${GGML_BUILD_DIR}/src/Release/ggml-cpu.lib" "${GGML_BUILD_DIR}/src/ggml-vulkan/Release/ggml-vulkan.lib")
if(MSVC)
  target_compile_options(kmd-keyframes PRIVATE /EHsc /W4)
endif()
